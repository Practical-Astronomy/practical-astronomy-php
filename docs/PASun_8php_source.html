30function approximate_position_of_sun($lctHours, $lctMinutes, $lctSeconds, $localDay, $localMonth, $localYear, $isDaylightSaving, $zoneCorrection)
31{
32 $daylightSaving = ($isDaylightSaving == true) ? 1 : 0;
33
34 $greenwichDateDay = PA_Macros\local_civil_time_greenwich_day($lctHours, $lctMinutes, $lctSeconds, $daylightSaving, $zoneCorrection, $localDay, $localMonth, $localYear);
35 $greenwichDateMonth = PA_Macros\local_civil_time_greenwich_month($lctHours, $lctMinutes, $lctSeconds, $daylightSaving, $zoneCorrection, $localDay, $localMonth, $localYear);
36 $greenwichDateYear = PA_Macros\local_civil_time_greenwich_year($lctHours, $lctMinutes, $lctSeconds, $daylightSaving, $zoneCorrection, $localDay, $localMonth, $localYear);
37 $utHours = PA_Macros\local_civil_time_to_universal_time($lctHours, $lctMinutes, $lctSeconds, $daylightSaving, $zoneCorrection, $localDay, $localMonth, $localYear);
38 $utDays = $utHours / 24;
39 $jdDays = PA_Macros\civil_date_to_julian_date($greenwichDateDay, $greenwichDateMonth, $greenwichDateYear) + $utDays;
40 $dDays = $jdDays - PA_Macros\civil_date_to_julian_date(0, 1, 2010);
41 $nDeg = 360 * $dDays / 365.242191;
42 $mDeg1 = $nDeg + PA_Macros\sun_e_long(0, 1, 2010) - PA_Macros\sun_peri(0, 1, 2010);
43 $mDeg2 = $mDeg1 - 360 * floor($mDeg1 / 360);
44 $eCDeg = 360 * PA_Macros\sun_ecc(0, 1, 2010) * sin(deg2rad($mDeg2)) / pi();
45 $lSDeg1 = $nDeg + $eCDeg + PA_Macros\sun_e_long(0, 1, 2010);
46 $lSDeg2 = $lSDeg1 - 360 * floor($lSDeg1 / 360);
47 $raDeg = PA_Macros\ec_ra($lSDeg2, 0, 0, 0, 0, 0, $greenwichDateDay, $greenwichDateMonth, $greenwichDateYear);
48 $raHours = PA_Macros\decimal_degrees_to_degree_hours($raDeg);
49 $decDeg = PA_Macros\ec_dec($lSDeg2, 0, 0, 0, 0, 0, $greenwichDateDay, $greenwichDateMonth, $greenwichDateYear);
50
51 $sunRAHour = PA_Macros\decimal_hours_hour($raHours);
52 $sunRAMin = PA_Macros\decimal_hours_minute($raHours);
53 $sunRASec = PA_Macros\decimal_hours_second($raHours);
54 $sunDecDeg = PA_Macros\decimal_degrees_degrees($decDeg);
55 $sunDecMin = PA_Macros\decimal_degrees_minutes($decDeg);
56 $sunDecSec = PA_Macros\decimal_degrees_seconds($decDeg);
57
58 return array($sunRAHour, $sunRAMin, $sunRASec, $sunDecDeg, $sunDecMin, $sunDecSec);

98 $eccentricity = PA_Macros\sun_ecc($gDay, $gMonth, $gYear);

102
103 $sunDistKm = round($rKm, 0);

173function equation_of_time($gwdateDay, $gwdateMonth, $gwdateYear)
