51function approximate_position_of_moon($lctHour, $lctMin, $lctSec, $isDaylightSaving, $zoneCorrectionHours, $localDateDay, $localDateMonth, $localDateYear)
52{
53 $daylightSaving = $isDaylightSaving ? 1 : 0;
54
55 $l0 = 91.9293359879052;
56 $p0 = 130.143076320618;
57 $n0 = 291.682546643194;
58 $i = 5.145396;
59
60 $gdateDay = local_civil_time_greenwich_day($lctHour, $lctMin, $lctSec, $daylightSaving, $zoneCorrectionHours, $localDateDay, $localDateMonth, $localDateYear);
61 $gdateMonth = local_civil_time_greenwich_month($lctHour, $lctMin, $lctSec, $daylightSaving, $zoneCorrectionHours, $localDateDay, $localDateMonth, $localDateYear);
62 $gdateYear = local_civil_time_greenwich_year($lctHour, $lctMin, $lctSec, $daylightSaving, $zoneCorrectionHours, $localDateDay, $localDateMonth, $localDateYear);
63
64 $utHours = local_civil_time_to_universal_time($lctHour, $lctMin, $lctSec, $daylightSaving, $zoneCorrectionHours, $localDateDay, $localDateMonth, $localDateYear);
65 $dDays = civil_date_to_julian_date($gdateDay, $gdateMonth, $gdateYear) - civil_date_to_julian_date(0.0, 1, 2010) + $utHours / 24;
66 $sunLongDeg = sun_long($lctHour, $lctMin, $lctSec, $daylightSaving, $zoneCorrectionHours, $localDateDay, $localDateMonth, $localDateYear);
67 $sunMeanAnomalyRad = sun_mean_anomaly($lctHour, $lctMin, $lctSec, $daylightSaving, $zoneCorrectionHours, $localDateDay, $localDateMonth, $localDateYear);
68 $lmDeg = unwind_deg(13.1763966 * $dDays + $l0);
69 $mmDeg = unwind_deg($lmDeg - 0.1114041 * $dDays - $p0);
70 $nDeg = unwind_deg($n0 - (0.0529539 * $dDays));
71 $evDeg = 1.2739 * sin(deg2rad(2.0 * ($lmDeg - $sunLongDeg) - $mmDeg));
72 $aeDeg = 0.1858 * sin($sunMeanAnomalyRad);
73 $a3Deg = 0.37 * sin($sunMeanAnomalyRad);
74 $mmdDeg = $mmDeg + $evDeg - $aeDeg - $a3Deg;
75 $ecDeg = 6.2886 * sin(deg2rad($mmdDeg));
76 $a4Deg = 0.214 * sin(2.0 * deg2rad($mmdDeg));
77 $ldDeg = $lmDeg + $evDeg + $ecDeg - $aeDeg + $a4Deg;
78 $vDeg = 0.6583 * sin(2.0 * deg2rad($ldDeg - $sunLongDeg));
79 $lddDeg = $ldDeg + $vDeg;
80 $ndDeg = $nDeg - 0.16 * sin($sunMeanAnomalyRad);
81 $y = sin(deg2rad($lddDeg - $ndDeg)) * cos(deg2rad($i));
82 $x = cos(deg2rad($lddDeg - $ndDeg));
83
84 $moonLongDeg = unwind_deg(w_to_degrees(atan2($y, $x)) + $ndDeg);
85 $moonLatDeg = w_to_degrees(asin(sin(deg2rad($lddDeg - $ndDeg)) * sin(deg2rad($i))));
86 $moonRAHours1 = decimal_degrees_to_degree_hours(ec_ra($moonLongDeg, 0, 0, $moonLatDeg, 0, 0, $gdateDay, $gdateMonth, $gdateYear));
87 $moonDecDeg1 = ec_dec($moonLongDeg, 0, 0, $moonLatDeg, 0, 0, $gdateDay, $gdateMonth, $gdateYear);
88
89 $moonRAHour = decimal_hours_hour($moonRAHours1);
90 $moonRAMin = decimal_hours_minute($moonRAHours1);
91 $moonRASec = decimal_hours_second($moonRAHours1);
92 $moonDecDeg = decimal_degrees_degrees($moonDecDeg1);
93 $moonDecMin = decimal_degrees_minutes($moonDecDeg1);
94 $moonDecSec = decimal_degrees_seconds($moonDecDeg1);
95
96 return array($moonRAHour, $moonRAMin, $moonRASec, $moonDecDeg, $moonDecMin, $moonDecSec);

130function moon_phase($lctHour, $lctMin, $lctSec, $isDaylightSaving, $zoneCorrectionHours, $localDateDay, $localDateMonth, $localDateYear, $accuracyLevel)
131{
132 $daylightSaving = $isDaylightSaving ? 1 : 0;
133
134 $gdateDay = local_civil_time_greenwich_day($lctHour, $lctMin, $lctSec, $daylightSaving, $zoneCorrectionHours, $localDateDay, $localDateMonth, $localDateYear);
135 $gdateMonth = local_civil_time_greenwich_month($lctHour, $lctMin, $lctSec, $daylightSaving, $zoneCorrectionHours, $localDateDay, $localDateMonth, $localDateYear);
136 $gdateYear = local_civil_time_greenwich_year($lctHour, $lctMin, $lctSec, $daylightSaving, $zoneCorrectionHours, $localDateDay, $localDateMonth, $localDateYear);
137
138 $sunLongDeg = sun_long($lctHour, $lctMin, $lctSec, $daylightSaving, $zoneCorrectionHours, $localDateDay, $localDateMonth, $localDateYear);
139 list($moonLongDeg, $moonLatDeg, $moonHorPara) = moon_long_lat_hp($lctHour, $lctMin, $lctSec, $daylightSaving, $zoneCorrectionHours, $localDateDay, $localDateMonth, $localDateYear);
140 $dRad = deg2rad($moonLongDeg - $sunLongDeg);
141
142 $moonPhase1 = ($accuracyLevel == AccuracyLevel::Precise)
143 ? moon_phase_ma($lctHour, $lctMin, $lctSec, $daylightSaving, $zoneCorrectionHours, $localDateDay, $localDateMonth, $localDateYear)
144 : (1.0 - cos($dRad)) / 2.0;
145
146 $sunRARad = deg2rad(ec_ra($sunLongDeg, 0, 0, 0, 0, 0, $gdateDay, $gdateMonth, $gdateYear));
147 $moonRARad = deg2rad(ec_ra($moonLongDeg, 0, 0, $moonLatDeg, 0, 0, $gdateDay, $gdateMonth, $gdateYear));
148 $sunDecRad = deg2rad(ec_dec($sunLongDeg, 0, 0, 0, 0, 0, $gdateDay, $gdateMonth, $gdateYear));
149 $moonDecRad = deg2rad(ec_dec($moonLongDeg, 0, 0, $moonLatDeg, 0, 0, $gdateDay, $gdateMonth, $gdateYear));
150
151 $y = cos($sunDecRad) * sin($sunRARad - $moonRARad);
152 $x = cos($moonDecRad) * sin($sunDecRad) - sin($moonDecRad) * cos($sunDecRad) * cos($sunRARad - $moonRARad);
153
154 $chiDeg = w_to_degrees(atan2($y, $x));
155
156 $moonPhase = round($moonPhase1, 2);
157 $paBrightLimbDeg = round($chiDeg, 2);
158
159 return array($moonPhase, $paBrightLimbDeg);
