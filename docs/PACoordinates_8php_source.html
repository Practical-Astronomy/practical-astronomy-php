446function heliographic_coordinates($helioPositionAngleDeg, $helioDisplacementArcmin, $gwdateDay, $gwdateMonth, $gwdateYear)
447{
448 $julianDateDays = PA_Macros\civil_date_to_julian_date($gwdateDay, $gwdateMonth, $gwdateYear);
449 $tCenturies = ($julianDateDays - 2415020) / 36525;
450 $longAscNodeDeg = PA_Macros\degrees_minutes_seconds_to_decimal_degrees(74, 22, 0) + (84 * $tCenturies / 60);
451 $sunLongDeg = PA_Macros\sun_long(0, 0, 0, 0, 0, $gwdateDay, $gwdateMonth, $gwdateYear);
452 $y = sin(deg2rad($longAscNodeDeg - $sunLongDeg)) * cos(deg2rad(PA_Macros\degrees_minutes_seconds_to_decimal_degrees(7, 15, 0)));
453 $x = -cos(deg2rad($longAscNodeDeg - $sunLongDeg));
454 $aDeg = PA_Macros\degrees(atan2($y, $x));
455 $mDeg1 = 360 - (360 * ($julianDateDays - 2398220) / 25.38);
456 $mDeg2 = $mDeg1 - 360 * floor($mDeg1 / 360);
457 $l0Deg1 = $mDeg2 + $aDeg;
458 $b0Rad = asin(sin(deg2rad($sunLongDeg - $longAscNodeDeg)) * sin(deg2rad(PA_Macros\degrees_minutes_seconds_to_decimal_degrees(7, 15, 0))));
459 $theta1Rad = atan(-cos(deg2rad($sunLongDeg)) * tan(deg2rad(PA_Macros\obliq($gwdateDay, $gwdateMonth, $gwdateYear))));
460 $theta2Rad = atan(-cos(deg2rad($longAscNodeDeg - $sunLongDeg)) * tan(deg2rad(PA_Macros\degrees_minutes_seconds_to_decimal_degrees(7, 15, 0))));
461 $pDeg = PA_Macros\degrees($theta1Rad + $theta2Rad);
462 $rho1Deg = $helioDisplacementArcmin / 60;
463 $rhoRad = asin(2 * $rho1Deg / PA_Macros\sun_dia(0, 0, 0, 0, 0, $gwdateDay, $gwdateMonth, $gwdateYear)) - deg2rad($rho1Deg);
464 $bRad = asin(sin($b0Rad) * cos($rhoRad) + cos($b0Rad) * sin($rhoRad) * cos(deg2rad($pDeg - $helioPositionAngleDeg)));
465 $bDeg = PA_Macros\degrees($bRad);
466 $lDeg1 = PA_Macros\degrees(asin(sin($rhoRad) * sin(deg2rad($pDeg - $helioPositionAngleDeg)) / cos($bRad))) + $l0Deg1;
467 $lDeg2 = $lDeg1 - 360 * floor($lDeg1 / 360);
468
469 $helioLongDeg = round($lDeg2, 2);
470 $helioLatDeg = round($bDeg, 2);
471
472 return array($helioLongDeg, $helioLatDeg);
